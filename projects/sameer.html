# build a variable label vector for better labelling of df
label_vec <- read_csv(filename) |>
  distinct(`Series Code`, `Series Name`) |>
  filter(!is.na(`Series Code`), !is.na(`Series Name`)) |>
  group_by(`Series Code`) |> # in case of duplicates
  summarise(label = first(`Series Name`), .groups = "drop") |>
  with(setNames(label, `Series Code`))

# compose dataframe
df <- read_csv(filename) |>
  select(-c(`Series Name`)) |>
  slice(1:(n() - 5)) |>
  pivot_longer(
    cols = matches("^\\d{4} \\[YR\\d{4}\\]$"),
    names_to = "year",
    values_to = "value",
    names_transform = list(year = ~ str_extract(.x, "^\\d{4}"))
  ) |>
  pivot_wider(names_from = `Series Code`, values_from = value) |>
  mutate(across(
    any_of(names(label_vec)),
    ~ {
      attr(.x, "label") <- unname(label_vec[cur_column()])
      .x
    }
  )) |>
  left_join(
    read_csv(filename2),
    join_by(`Country Code` == Code)
  )

df <- df |>
  mutate(
    SM.POP.NETM.prop = (SM.POP.NETM / SP.POP.TOTL) * 100
  )

plots <- variables |>
  keep(~ .x %in% names(df) && is.numeric(df[[.x]])) |>
  set_names() |>
  map(
    \(v) {
      x_lab <- attr(df[[v]], "label") %||% v

      x_vals <- df[[v]]
      x_vals_non_na <- x_vals[!is.na(x_vals)]

      # Summary stats
      mean_val <- mean(x_vals_non_na)
      sd_val <- sd(x_vals_non_na)

      nepal_value <- df |>
        filter(`Country Name` == "Nepal") |>
        pull(.data[[v]]) |>
        first()

      # Detect whether values are effectively integers
      is_integerish <- length(x_vals_non_na) > 0 &&
        all(abs(x_vals_non_na - round(x_vals_non_na)) < 1e-8)

      # Choose histogram settings automatically
      if (is_integerish && diff(range(x_vals_non_na)) <= 50) {
        hist_geom <- geom_histogram(
          binwidth = 1,
          boundary = -0.5,
          color = "gray40",
          fill = "gray80",
          na.rm = TRUE
        )
      } else {
        # Freedman-Diaconis rule for continuous variables
        iqr_val <- IQR(x_vals_non_na)
        n_val <- length(x_vals_non_na)

        fd_binwidth <- 2 * iqr_val / (n_val^(1 / 3))
        fd_binwidth <- 0.5 * fd_binwidth

        # Fallback if IQR is zero or binwidth is not usable
        if (!is.finite(fd_binwidth) || fd_binwidth <= 0) {
          rng <- diff(range(x_vals_non_na))
          fd_binwidth <- if (is.finite(rng) && rng > 0) rng / 30 else 1
        }

        hist_geom <- geom_histogram(
          binwidth = fd_binwidth,
          color = "gray40",
          fill = "gray80",
          na.rm = TRUE
        )
      }

      caption_text <- paste0(
        "Mean = ",
        scales::number(mean_val, accuracy = 0.01),
        "   |   SD = ",
        scales::number(sd_val, accuracy = 0.01)
      )

      p <- ggplot(df, aes(x = .data[[v]])) +
        hist_geom +
        annotate(
          "rect",
          xmin = mean_val - sd_val,
          xmax = mean_val + sd_val,
          ymin = -Inf,
          ymax = Inf,
          fill = "blue",
          alpha = 0.12
        ) +
        geom_vline(
          xintercept = mean_val,
          color = "blue",
          linewidth = 0.9,
          alpha = 0.48
        ) +
        # geom_vline(
        #   xintercept = c(mean_val - sd_val, mean_val + sd_val),
        #   color = "blue",
        #   linetype = "dashed",
        #   linewidth = 0.8
        # ) +
        labs(
          title = paste("Histogram of", x_lab),
          x = x_lab,
          y = "Count",
          caption = caption_text
        ) +
        theme_minimal() +
        theme(
          plot.caption = element_text(
            hjust = 0.5,
            margin = margin(t = 8)
          ),
          plot.caption.position = "plot"
        )

      if (!is.na(nepal_value)) {
        p <- p +
          geom_vline(
            xintercept = nepal_value,
            color = "red",
            linetype = "dotted",
            linewidth = 0.8
          )
      }

      p
    }
  )

\[ \begin{align} VE_i &\sim \text{Normal}(\mu_i,\sigma) \\ \mu_i &= \alpha + \beta_{CED} CED_i + \beta_{YN} YN_i + \beta_{RL} RL_i + \beta_{VA} VA_i \\ \alpha &\sim \text{Normal}(50,25) \\ \beta_{CED} &\sim \text{Normal}(0,10) \\ \beta_{YN} &\sim \text{Normal}(0,10) \\ \beta_{RL} &\sim \text{Normal}(0,10) \\ \beta_{VA} &\sim \text{Normal}(0,10) \\ \sigma &\sim \text{Normal}(0,20), \quad \sigma > 0 \end{align} \]

Variable Description Unit
\(VE_i\) Vulnerable Employment percentage
\(CED_i\) Compulsory Education Duration standardized
\(YN_i\) Youth not in Education, Employment, or Training standardized
\(RL_i\) Rule of Law standardized
\(VA_i\) Voice and Accountability standardized
